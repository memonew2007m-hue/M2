❤️`; // 1. إرسال البيانات تلقائياً إلى Google Sheets const sheetData = { orderType: typeName, tableDetails: tableDetails, items: itemsSummaryList.join(" | "), deliveryFee: deliveryFee, grandTotal: grandTotal, notes: orderNotes || "بدون ملاحظات" }; fetch(googleAppScriptUrl, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify(sheetData) }).catch(err => console.error("خطأ أثناء تسجيل الطلب في الشيت:", err)); // 2. تحويل الزبون إلى واتساب window.open(`https://wa.me/${phoneNumber}?text=${encodeURIComponent(message)}`, '_blank'); }
