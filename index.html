function sendOrderToWhatsApp() { const customerName = document.getElementById('customerName').value.trim(); const customerPhone = document.getElementById('customerPhone').value.trim(); const orderType = document.getElementById('orderType').value; const deliveryZoneSelect = document.getElementById('deliveryZone'); const deliveryZoneText = deliveryZoneSelect.options[deliveryZoneSelect.selectedIndex]?.text || ''; const tableNumber = document.getElementById('tableNumber').value.trim(); const orderNotes = document.getElementById('orderNotes').value.trim(); if (!customerName || !customerPhone) { alert('يرجى ملا الاسم ورقم الهاتف!'); return; } if (cart.length === 0) { alert('السلة فارغة!'); return; } // 1. تجهيز نص الطلب للواتساب let orderDetailsText = ""; cart.forEach((item, index) => { orderDetailsText += `${index + 1}. ${item.name} (عدد: ${item.quantity}) - ${(item.price * item.quantity).toLocaleString()} د.ع\n`; }); const subtotal = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0); let deliveryFee = 0; let addressOrTable = ""; if (orderType === 'delivery') { deliveryFee = parseInt(deliveryZoneSelect.value) || 0; addressOrTable = `التوصيل إلى: ${deliveryZoneText}`; } else if (orderType === 'hall') { addressOrTable = `رقم الطاولة: ${tableNumber || 'غير محدد'}`; } else { addressOrTable = "سفري (استلام من المطعم)"; } const grandTotal = subtotal + deliveryFee; // 2. إرسال البيانات إلى Google Sheets في الخلفية const sheetData = { timestamp: new Date().toLocaleString('ar-IQ'), name: customerName, phone: customerPhone, orderType: orderType === 'delivery' ? 'توصيل' : (orderType === 'hall' ? 'صالة' : 'سفري'), address: addressOrTable, items: cart.map(i => `${i.name} (x${i.quantity})`).join(', '), total: grandTotal.toLocaleString() + ' د.ع', notes: orderNotes || 'لا يوجد' }; const googleScriptUrl = 'https://script.google.com/macros/s/AKfycbxWcF-w58PN01Nx7-B7C7pukSNoF7C231Hj3CRfxyc3XLoS5Z2U-o2a2QAHMGKmNvZ0/exec'; fetch(googleScriptUrl, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(sheetData) }).catch(err => console.error('Sheet error:', err)); // 3. فتح الواتساب ومسح السلة كالمعتاد let message = `*طلب جديد من مطعم سمسمه*\n\n`; message += `*اسم الزبون:* ${customerName}\n`; message += `*رقم الهاتف:* ${customerPhone}\n`; message += `*نوع الطلب:* ${sheetData.orderType}\n`; if (addressOrTable) message += `* التفاصيل:* ${addressOrTable}\n`; message += `\n*الأطباق:\n*${orderDetailsText}`; if (deliveryFee > 0) message += `\n*أجور التوصيل:* ${deliveryFee.toLocaleString()} د.ع`; message += `\n*المجموع الكلي:* ${grandTotal.toLocaleString()} د.ع\n`; if (orderNotes) message += `\n*ملاحظات:* ${orderNotes}\n`; const whatsappNumber = "9647723681360"; const encodedMessage = encodeURIComponent(message); // مسح السلة وإغلاق النافذة cart = []; updateCartUI(); closeCartModal(); // التوجيه للواتساب window.open(`https://wa.me/${whatsappNumber}?text=${encodedMessage}`, '_blank'); }
